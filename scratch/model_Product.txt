model Product {
  id              String   @id @default(auto()) @map("_id") @db.ObjectId
  name            String
  description     String?
  longDescription String? // Added: From General Details section
  category        String? // Consider making this an enum or relation if fixed categories
  subCategory     Json? // Consider a specific model if structure is fixed
  subCategoryName String? // Added: From General Details
  images          Json[] // Store URLs or paths
  videos          Json[]
  ebooks          Json[]
  tags            String[] // Array of strings for search/filtering
  brand           String?

  // Company relation
  companyId String?  @db.ObjectId
  company   Company? @relation(fields: [companyId], references: [id])

  // Product Category relation
  productCategoryId String?          @db.ObjectId
  productCategory   ProductCategory? @relation(fields: [productCategoryId], references: [id])

  // Inventory & Request Management
  inventoryItems      InventoryItem[]
  requests            Request[]             @relation("ProductRequests")
  commissions         Commission[]
  Target              Target[]
  CommissionRate      CommissionRate?
  marketplaceListings marketplaceListings[]
  socialCampaigns     SocialCampaign[]
  socialPosts         SocialMediaPost[]

  // General Specifications (from General Details)
  model      String?
  color      String[] // Array of colors (e.g., ["Red", "Blue"])
  size       String[] // Array of sizes (e.g., ["S", "M", "L"])
  weight     String[] // e.g., "5 kg"
  condition  String? // e.g., "New", "Used", "Refurbished"
  dimensions String? // e.g., "10x5x2 cm" - Renamed from 'dimension' for consistency
  material   String[] // Array for multiple materials, or a single String for one

  // Pricing & Quantity
  quantity     Int    @default(1) // Default to 1
  costPrice    Float  @default(0)
  sellingPrice Float  @default(0)
  discount     Int    @default(0) // Percentage (0-100)
  finalPrice   Float  @default(0) // Price after discount
  profitMargin Float  @default(0) // Calculated profit margin
  pricingTiers Json[] // Added: For complex pricing structures (e.g., [{ quantity: 5, price: 9.99 }])

  // Deal Specifics
  isOnOffer    Boolean @default(false)
  isFlashDeal  Boolean @default(false)
  isDiscounted Boolean @default(false)

  startDealDate DateTime?
  endDealDate   DateTime?

  // Flags
  isAvailable  Boolean @default(true)
  isNewArrival Boolean @default(false)
  isFeatured   Boolean @default(false)

  // Vehicle Specifics (Engine & Performance, Ownership & Pricing)
  make String? // e.g., "Toyota", "Ford"
  trim String? // e.g., "Sport", "Luxury", "Base"
  type String? // e.g., "SUV", "Sedan", "Truck" // Added: From General Details

  mileage    String? // e.g., "50000 km"
  engineType String? // e.g., "V6", "Inline-4"
  engineSize Float? // e.g., 3.5 (for 3.5L) - Changed to Float
  horsepower Int? // Added: Horsepower (e.g., 250) - Changed to Int

  torque       Int? // Added: Torque (e.g., 280) - Changed to Int
  fuelType     String? // Added: e.g., "Petrol", "Electric"
  fuelEconomy  String? // Added: e.g., "7.5 L/100km", "32 MPG"
  transmission String? // e.g., "Automatic", "Manual"
  drivetrain   String? // e.g., "FWD", "RWD", "AWD"

  vin            String? // Vehicle Identification Number
  logbookStatus  String? // e.g., "Available", "Missing"
  serviceHistory String? // e.g., "Full", "Partial", "None"
  negotiable     Boolean @default(false) // Pricing option

  financingAvailable Boolean @default(false) // Sales feature
  tradeIn            Boolean @default(false) // Sales feature
  features           Json[] // Array of miscellaneous vehicle features (e.g., safety, tech)

  // New Ownership & Pricing fields
  previousOwners    Int? // Added: Number of previous owners - Changed to Int
  tireCondition     String? // Added: e.g., "New", "Good", "Fair", "Poor"
  accidentalHistory Boolean? @default(false) // Added: Has reported accidental history

  // Books Specifics
  author    String?
  publisher String?
  isbn      String?

  // Fashion Specifics
  fabricComposition String?
  careInstructions  String?

  // Home Appliances Specifics
  energyRating        String?
  warrantyPeriod      String?
  applianceDimensions String? // To avoid conflict, renamed from 'dimensions'

  // Beauty Products Specifics
  ingredients       String?
  usageInstructions String?
  expirationDate    DateTime?

  // Options & Amenities
  // Stores structured items matching: { category: string, name: string, extraPrice: number }
  // option            Json[]
  option    Json[] // Generic options (e.g., for configurable products)
  amenities String[] // Specific for properties/services

  // Property Specifics
  propertyType   PropertyType? @relation(fields: [propertyTypeId], references: [id])
  propertyTypeId String?       @db.ObjectId

  area      String? // e.g., "1500 sqft"
  bedrooms  Json[] // Updated to support multiple bedroom types
  studios   Json[] // Updated to support multiple bedroom types
  bathrooms String?

  // Service & Scheduling Specifics
  serviceSchedule String?
  year            Int? // Changed to Int (e.g., vehicle model year, or year of service)

  availabilityStart DateTime? // For services/rentals
  availabilityEnd   DateTime? // For services/rentals

  // Location & Contact
  location     Json? // GeoJSON format { type: "Point", coordinates: [lng, lat] }
  locationName String?
  latitude     Float?
  longitude    Float?

  contact     String? // General contact number
  contactName String?
  email       String?

  // Marketplace & Fulfillment
  delivery          Boolean  @default(false)
  paymentOption     String   @default("AT SHOP")
  showOnGhuba       Boolean? @default(true)
  digitalUrl        String?
  autoDeliver       Boolean? @default(false)
  hourlyRate        Float?
  minimumHours      Int?
  minNoticePeriod   String?
  maxBookingAhead   String?
  totalCapacity     Int?
  deliveryMethod    String?
  fulfillmentStatus String? // Consider an enum
  providerRating    Float? // e.g., 4.5 stars
  bookingSlots      Json[] // Store complex booking slot data as JSON

  // Admin & Auditing
  status ListingStatus @default(ACTIVE) // Assuming ListingStatus enum exists

  listingMarketStatus    ListingMarketStatus    @default(AVAILABLE)
  listingSystemStatus    ListingSystemStatus    @default(DRAFT)
  listingTransactionType ListingTransactionType @default(SALE)

  //buy, rent, rented, draft
  active Boolean @default(true)

  Collection     Collection?      @relation(fields: [collectionId], references: [id])
  collectionId   String?          @db.ObjectId
  CollectionItem CollectionItem[]
  Review         Review[]
  Question       Question[]
  RecentlyViewed RecentlyViewed[]
  WishlistItem   WishlistItem[]
  ProductMetrics ProductMetrics[]

  createdAt DateTime @default(now())
  updatedAt DateTime @updatedAt

  // Additional fields from original schema, check if still relevant or duplicated
  tax          Float? @default(0)
  shippingCost Float? @default(0)

  // Relations that might need to be defined elsewhere in schema.prisma
  myLocation Location?   @relation(fields: [locationId], references: [id], onDelete: NoAction, onUpdate: NoAction)
  locationId String?     @db.ObjectId
  OrderItem  OrderItem[]

  // Indexes for performance
  @@index([name])
  @@index([finalPrice])
  @@index([isFeatured])
  @@index([createdAt])
  @@index([status])
  @@index([companyId]) // Good for fetching products by company
  @@index([productCategoryId]) // Good for filtering by category
  @@index([propertyTypeId])
  @@index([collectionId])
  @@index([locationId])
  @@index([tags])
  @@index([images])
  @@index([email])
  @@index([companyId, status, active])
  @@index([companyId, createdAt])
  @@index([productCategoryId, status, active])
  @@index([isFeatured, status])
  @@index([isNewArrival, createdAt])
  @@index([listingMarketStatus, active])
  @@index([listingSystemStatus])
  @@index([companyId, listingTransactionType, isFeatured, createdAt])
  @@index([companyId, listingTransactionType, finalPrice])
  @@index([createdAt, id]) // Required for efficient Cursor-based pagination
  @@index([isFeatured, createdAt])
}
