model OrderItem {
  id                   String               @id @default(auto()) @map("_id") @db.ObjectId
  orderId              String               @db.ObjectId
  order                CustomerOrder        @relation(fields: [orderId], references: [id])
  marketplaceListingId String?              @db.ObjectId
  marketplaceListing   marketplaceListings? @relation(fields: [marketplaceListingId], references: [id])
  quantity             Int
  price                Float
  totalPrice           Float?
  discount             Float?
  tax                  Float?
  serviceNotes         String?
  course               String? // e.g., "Starters", "Mains", "Drinks", "Desserts"
  kitchenStatus        String?              @default("PENDING")
  voidReason           String?
  voidedByStaff        String?
  selectedOptions      Json? // e.g., [{category: "color", name: "Red", extraPrice: 20}, {category: "size", name: "7", extraPrice: 50}]

  riderId String?
  status  OrderStatus? @default(PENDING)

  date     String?
  timeSlot String? // e.g. "09:00"

  Return Return[]

  productId String?  @db.ObjectId
  product   Product? @relation(fields: [productId], references: [id])

  appointmentId String?      @db.ObjectId
  appointment   Appointment? @relation(fields: [appointmentId], references: [id])

  createdAt DateTime? @default(now())
  updatedAt DateTime? @updatedAt

  @@index([productId])
  @@index([appointmentId])
  @@index([marketplaceListingId])
  @@index([orderId])
  @@index([status])
  @@index([marketplaceListingId, createdAt])
  @@index([orderId, marketplaceListingId])
}
