// Conversation Model
// Represents a message thread or chat between multiple participants.
model Conversation {
  id            String    @id @default(auto()) @map("_id") @db.ObjectId
  title         String? // Optional: For group chats, or subject of a direct conversation
  companyId     String    @db.ObjectId // For multi-tenancy: Ensures conversations belong to a specific company
  company       Company   @relation(fields: [companyId], references: [id])
  createdAt     DateTime  @default(now())
  updatedAt     DateTime  @updatedAt // Automatically updates on any change to the conversation or its messages
  lastMessageAt DateTime? // Stores the timestamp of the most recent message for easy sorting in inbox views

  // Relations
  participants ConversationParticipant[] // Links to users participating in this conversation
  messages     Message[] // Links to individual messages within this conversation

  // Indexes for efficient querying
  @@index([companyId])
  @@index([updatedAt]) // Useful for sorting conversations by recent activity
}

// ConversationParticipant Model
// This is a join table that links a User to a Conversation.
// It also stores individual user-specific settings for that conversation,
// such as read status and soft deletion/archiving.
model ConversationParticipant {
  id             String       @id @default(auto()) @map("_id") @db.ObjectId
  conversationId String       @db.ObjectId
  conversation   Conversation @relation(fields: [conversationId], references: [id], onDelete: Cascade) // If conversation is deleted, so are its participants

  userId String @db.ObjectId
  user   User   @relation(fields: [userId], references: [id])

  // Individual status for this participant in this conversation
  isArchived Boolean @default(false) // True if the user has archived this conversation from their inbox
  isDeleted  Boolean @default(false) // True if the user has "deleted" this conversation from their view (soft delete)

  // Tracks the last message ID this user has read in this conversation.
  // This is efficient for calculating unread messages in a conversation.
  lastReadMessageId String? @db.ObjectId

  // Denormalized count of unread messages for quick display in inbox list.
  // This would need to be updated via application logic (e.g., a background job or on message send/read).
  unreadCount Int @default(0)

  createdAt DateTime @default(now())
  updatedAt DateTime @updatedAt

  // Ensures a user can only be a participant in a specific conversation once
  @@unique([conversationId, userId])
  // Indexes for efficient querying
  @@index([userId])
  @@index([conversationId])
}

// Message Model
// Represents an individual message sent within a conversation.
model Message {
  id             String       @id @default(auto()) @map("_id") @db.ObjectId
  conversationId String       @db.ObjectId
  conversation   Conversation @relation(fields: [conversationId], references: [id], onDelete: Cascade) // If conversation is deleted, its messages are deleted

  senderId String @db.ObjectId
  sender   User   @relation("SentMessages", fields: [senderId], references: [id]) // Renamed relation for clarity

  content        String // The actual message text
  messageType    MessageType @default(TEXT) // Categorizes the content (e.g., text, image, file)
  attachmentUrls String[] // Stores URLs to any attached files or images

  createdAt DateTime @default(now()) // Timestamp when the message was sent

  // Indexes for efficient querying
  @@index([conversationId])
  @@index([senderId])
  @@index([createdAt]) // Essential for ordering messages chronologically within a conversation
}

  MULTIPLE_CHOICE
  TRUE_FALSE
  SHORT_ANSWER
  ESSAY
  FILL_IN_THE_BLANK
  MATCHING
  NUMERIC
}

// Enum for Course Material Type
enum CourseMaterialType {
  DOCUMENT // PDF, DOC, PPT
  VIDEO
  LINK // External website
  IMAGE
  AUDIO
  OTHER
}

// Course Material Model
// Stores links to various educational resources for a course.
model CourseMaterial {
  id           String             @id @default(auto()) @map("_id") @db.ObjectId
  courseId     String             @db.ObjectId
  course       Course             @relation(fields: [courseId], references: [id])
  title        String
  description  String?
  fileUrl      String? // URL for downloadable files (PDFs, docs)
  linkUrl      String? // URL for external links (videos, websites)
  type         CourseMaterialType @default(DOCUMENT) // Enum: DOCUMENT, VIDEO, LINK, IMAGE, AUDIO
  uploadedById String             @db.ObjectId // Educator who uploaded the material
  uploadedBy   Educator           @relation("UploadedMaterial", fields: [uploadedById], references: [id])
  createdAt    DateTime           @default(now())
  updatedAt    DateTime           @updatedAt
  Educator     Educator?          @relation(fields: [educatorId], references: [id])
  educatorId   String?            @db.ObjectId

  @@index([courseId, type])
  @@index([uploadedById])
}

// Discussion Topic Model
// Represents a discussion forum topic within a course or globally.
model DiscussionTopic {
  id          String   @id @default(auto()) @map("_id") @db.ObjectId
  courseId    String?  @db.ObjectId // Optional: if topic is specific to a course
  course      Course?  @relation(fields: [courseId], references: [id])
  title       String
  content     String? // Initial post content
  createdById String   @db.ObjectId // User (Educator or Student) who created the topic
  createdBy   Educator @relation("CreatedDiscussionTopic", fields: [createdById], references: [id])

  posts     DiscussionPost[] // All posts within this topic
  createdAt DateTime         @default(now())
  updatedAt DateTime         @updatedAt
  User      User?            @relation(fields: [userId], references: [id])
  userId    String?          @db.ObjectId

  @@index([courseId])
  @@index([createdById])
}

// Discussion Post Model
// Represents an individual post or reply within a discussion topic.
model DiscussionPost {
  id          String           @id @default(auto()) @map("_id") @db.ObjectId
  topicId     String           @db.ObjectId
  topic       DiscussionTopic  @relation(fields: [topicId], references: [id])
  parentId    String?          @db.ObjectId // Optional: For replies to other posts
  parent      DiscussionPost?  @relation("ReplyToPost", fields: [parentId], references: [id], onDelete: NoAction, onUpdate: NoAction) // Self-referencing
  content     String
  createdById String           @db.ObjectId // User (Educator or Student) who created the post
  createdBy   User             @relation("CreatedDiscussionPost", fields: [createdById], references: [id])
  replies     DiscussionPost[] @relation("ReplyToPost") // Replies to this post
  createdAt   DateTime         @default(now())
  updatedAt   DateTime         @updatedAt
  User        User?            @relation(fields: [userId], references: [id])
  userId      String?          @db.ObjectId

  @@index([topicId])
  @@index([parentId])
  @@index([createdById])
}

// Department Model
// Represents an academic or administrative department within the school.
model Department {
  id          String  @id @default(auto()) @map("_id") @db.ObjectId
  name        String // e.g., "Mathematics", "English", "Administration"
  description String?

  headId String?   @db.ObjectId // Optional: Link to the User (Educator/Admin) who heads the department
  head   Educator? @relation("DepartmentHead", fields: [headId], references: [id], onDelete: NoAction, onUpdate: NoAction)

  educators Educator[] // Educators belonging to this department
  courses   Course[] // Courses offered by this department
  createdAt DateTime   @default(now())
  updatedAt DateTime   @updatedAt
  Company   Company?   @relation(fields: [companyId], references: [id])
  companyId String?    @db.ObjectId
  User      User?      @relation(fields: [userId], references: [id])
  userId    String?    @db.ObjectId

  @@index([name])
}
